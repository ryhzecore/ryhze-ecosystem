# Allocate test evidence when CTest runs, rather than fixing a configure-time path.
# Executable targets must still resolve to their configuration-specific binaries.
function(race_add_test)
  cmake_parse_arguments(PARSE_ARGV 0 RACE_TEST "" "NAME" "COMMAND")
  if(NOT RACE_TEST_NAME OR NOT RACE_TEST_COMMAND OR RACE_TEST_UNPARSED_ARGUMENTS)
    message(FATAL_ERROR "race_add_test requires NAME and COMMAND")
  endif()
  # Current tests use ordinary nonempty arguments. Inspect original ARGV so an
  # explicit trailing empty argument cannot disappear during list forwarding.
  set(RACE_TEST_IN_COMMAND FALSE)
  math(EXPR RACE_TEST_LAST_ARGUMENT "${ARGC} - 1")
  foreach(RACE_TEST_ARGUMENT_INDEX RANGE 0 ${RACE_TEST_LAST_ARGUMENT})
    set(RACE_TEST_ARGUMENT "${ARGV${RACE_TEST_ARGUMENT_INDEX}}")
    if(RACE_TEST_IN_COMMAND AND (RACE_TEST_ARGUMENT STREQUAL "" OR RACE_TEST_ARGUMENT MATCHES ";"))
      message(FATAL_ERROR "race_add_test(${RACE_TEST_NAME}): empty or semicolon command arguments are not supported")
    elseif(RACE_TEST_ARGUMENT STREQUAL "COMMAND")
      set(RACE_TEST_IN_COMMAND TRUE)
    endif()
  endforeach()
  list(POP_FRONT RACE_TEST_COMMAND RACE_TEST_EXECUTABLE)
  set(RACE_TEST_LAUNCHER "")
  set(RACE_TEST_EMULATOR "")
  if(TARGET "${RACE_TEST_EXECUTABLE}")
    set(RACE_TEST_TARGET "${RACE_TEST_EXECUTABLE}")
    # Match add_test's launcher -> emulator -> executable order. The evidence
    # wrapper decodes each prefix independently without expanding child args.
    if(CMAKE_VERSION VERSION_GREATER_EQUAL "3.29")
      set(RACE_TEST_LAUNCHER
        "$<TARGET_GENEX_EVAL:${RACE_TEST_TARGET},$<TARGET_PROPERTY:${RACE_TEST_TARGET},TEST_LAUNCHER>>")
    endif()
    set(RACE_TEST_EMULATOR_POLICY "OLD")
    if(POLICY CMP0158)
      cmake_policy(GET CMP0158 RACE_TEST_EMULATOR_POLICY)
    endif()
    if(CMAKE_CROSSCOMPILING OR NOT RACE_TEST_EMULATOR_POLICY STREQUAL "NEW")
      if(CMAKE_VERSION VERSION_GREATER_EQUAL "3.29")
        set(RACE_TEST_EMULATOR
          "$<TARGET_GENEX_EVAL:${RACE_TEST_TARGET},$<TARGET_PROPERTY:${RACE_TEST_TARGET},CROSSCOMPILING_EMULATOR>>")
      else()
        get_target_property(RACE_TEST_EMULATOR "${RACE_TEST_TARGET}" CROSSCOMPILING_EMULATOR)
        if(NOT RACE_TEST_EMULATOR)
          set(RACE_TEST_EMULATOR "")
        endif()
      endif()
    endif()
    set(RACE_TEST_EXECUTABLE "$<TARGET_FILE:${RACE_TEST_TARGET}>")
  endif()
  add_test(NAME "${RACE_TEST_NAME}" COMMAND "${Python3_EXECUTABLE}" -B
    "${CMAKE_SOURCE_DIR}/scripts/run_evidence.py" --run "${RACE_TEST_NAME}"
    --cmake-launcher "${RACE_TEST_LAUNCHER}" --cmake-emulator "${RACE_TEST_EMULATOR}" --
    "${RACE_TEST_EXECUTABLE}" ${RACE_TEST_COMMAND})
endfunction()
